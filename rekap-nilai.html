async function tampilkanRekapNilaiOnline() { let filterKelas = document.getElementById("filter-kelas").value.trim(); let filterMapel = document.getElementById("filter-mapel").value.trim(); let tbody = document.getElementById("tabel-nilai-body"); tbody.innerHTML = `⏳ Mengambil rekap nilai dari Google Sheets...`; try { let response = await fetch(SCRIPT_URL + "?action=getNilai"); let dataNilai = await response.json(); tbody.innerHTML = ""; if (!Array.isArray(dataNilai) || dataNilai.length === 0) { tbody.innerHTML = `Belum ada data nilai di server.`; return; } let rekapMap = {}; dataNilai.forEach(item => { // Ambil data dengan berbagai kemungkinan nama properti (antisipasi huruf besar/kecil) let kls = String(item.kelas || item.Kelas || item.KELAS || "").trim(); let nms = String(item.nama || item["Nama Siswa"] || item.Nama || item.NAMA || "").trim(); let mpl = String(item.mapel || item["Mata Pelajaran"] || item.Mapel || item.MAPEL || "").trim(); if (!nms) return; // Abaikan baris kosong // Filter berdasarkan pilihan guru let matchKelas = filterKelas ? kls.toLowerCase() === filterKelas.toLowerCase() : true; let matchMapel = filterMapel ? mpl.toLowerCase() === filterMapel.toLowerCase() : true; if (matchKelas && matchMapel) { let key = `${kls}_${nms}_${mpl}`; if (!rekapMap[key]) { rekapMap[key] = { kelas: kls, nama: nms, mapel: mpl, "Pilihan Ganda": null, "Puzzle Kata": null, "Essay": null, "Merangkum Materi": null }; } let jenisSoal = String(item.jenis || item["Jenis Kuis"] || item.Jenis || item.JENIS || "").trim(); let valNilai = item.nilai !== undefined ? item.nilai : item.Nilai; // Petetapan jenis soal ke kolom rekap yang sesuai if (jenisSoal.toLowerCase().includes("pilihan ganda")) { rekapMap[key]["Pilihan Ganda"] = valNilai; } else if (jenisSoal.toLowerCase().includes("puzzle")) { rekapMap[key]["Puzzle Kata"] = valNilai; } else if (jenisSoal.toLowerCase().includes("essay")) { rekapMap[key]["Essay"] = valNilai; } else if (jenisSoal.toLowerCase().includes("merangkum")) { rekapMap[key]["Merangkum Materi"] = valNilai; } else if (rekapMap[key].hasOwnProperty(jenisSoal)) { rekapMap[key][jenisSoal] = valNilai; } } }); let keys = Object.keys(rekapMap); if (keys.length === 0) { tbody.innerHTML = `Tidak ada data yang cocok dengan filter yang dipilih.`; return; } let index = 1; keys.forEach(key => { let row = rekapMap[key]; let totalNilai = 0; let jumlahDikerjakan = 0; let jenisKuisList = ["Pilihan Ganda", "Puzzle Kata", "Essay", "Merangkum Materi"]; jenisKuisList.forEach(jenis => { let val = row[jenis]; if (val !== null && val !== undefined && val !== "") { let num = Number(val); if (!isNaN(num)) { totalNilai += num; jumlahDikerjakan++; } } }); let nilaiAkhir = jumlahDikerjakan > 0 ? Math.round(totalNilai / jumlahDikerjakan) : 0; let badgeColor = nilaiAkhir >= 75 ? 'success' : (nilaiAkhir >= 60 ? 'warning' : 'danger'); let renderKolom = (val) => { if (val === null || val === undefined || val === "") { return 'Belum Dikerjakan'; } let num = Number(val); if (isNaN(num)) { // Untuk teks seperti "Menunggu Koreksi" return `${val}`; } return `${val}`; }; let tr = document.createElement("tr"); tr.innerHTML = ` ${index++} ${row.kelas} ${row.nama} ${row.mapel} ${renderKolom(row["Pilihan Ganda"])} ${renderKolom(row["Puzzle Kata"])} ${renderKolom(row["Essay"])} ${renderKolom(row["Merangkum Materi"])} ${jumlahDikerjakan > 0 ? `${nilaiAkhir}` : '-'} `; tbody.appendChild(tr); }); } catch (e) { console.error(e); tbody.innerHTML = `Gagal memuat rekap nilai dari server.`; } }
